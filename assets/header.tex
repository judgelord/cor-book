  \ifstrequal{#1}{lof}{\stepcounter{totalfigures}}{}%

  \ifstrequal{#1}{apt}{\stepcounter{totaltables}}{}%
  \RunningTotalsOriginalAddcontentsline{#1}{#2}{#3}%
}
